# ---------------------------------------------------------------------------
# Runtime dependencies -- this is the file Streamlit Community Cloud installs.
# Training-only extras live in requirements-dev.txt so they never reach deploy.
#
# Every pin below was resolved and imported successfully by
#   scripts/verify_env.py
# on Python 3.11.15 / macOS 26.4.1 arm64. See PROGRESS.md for the reasoning.
# ---------------------------------------------------------------------------

# Torch CPU builds. The default PyPI Linux wheel bundles CUDA and is 900 MB,
# which blows the Streamlit Cloud image; the +cpu build from PyTorch's own
# index is a fraction of that. macOS has no +cpu variant (and needs the
# regular wheel for MPS), so the two platforms are pinned separately.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.9.1+cpu ; sys_platform == "linux"
torch==2.9.1 ; sys_platform == "darwin"

# Keras 3 runs on the torch backend -- set KERAS_BACKEND=torch before importing.
# TensorFlow is deliberately NOT a dependency: its native extension aborts on
# import on macOS 26 ("mutex lock failed"), on both 2.20.0 and 2.21.0.
keras==3.15.1

# Headless OpenCV: the regular opencv-python needs libGL.so.1, which is absent
# from the Streamlit Cloud base image and fails at import with an ImportError.
opencv-python-headless==4.12.0.88

numpy==2.2.6

# REQUIRED to unpickle svd.pkl and scaler.pkl. Its absence from the original
# requirements.txt meant inference could never have worked on a clean install.
scikit-learn==1.7.2
joblib==1.5.2

# Semantic search over generated captions. Static embeddings, numpy-only
# inference -- replaces sentence-transformers, which would have pulled a second
# copy of the transformers/torch stack.
model2vec==0.8.2

streamlit==1.51.0
